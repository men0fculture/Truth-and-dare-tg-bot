	$("#capimg").attr("src", "jsp/common/captchaimage.jsp?"+(new Date().getTime()));

function validateSubmit(){
	var applicationNumber = document.forms["authentication"]["llappln"] ;
	//var entcapt=document.getElementById('entcaptxt').value;
	var entcaptxt = $("#entcaptxt").val();
	sessionStorage.setItem('entcaptxt', entcaptxt);
	
	if (applicationNumber.value.trim() == "") {
		alert('Please Enter Application Number.');
		applicationNumber.focus();
		return false;
	}
	else if(entcaptxt.trim() == "") {
		alert("Please Enter Captcha.");
		document.getElementById('entcaptxt').focus();
		return false;
	}
	else
	{
		try {
						
			$.ajax({
				type : 'POST',
				url : 'getcaptcha.do',
				data : {
					
					entcaptxt:entcaptxt,
					llappln:applicationNumber.value.trim()

				},
				cache : false,
				async : false,
				dataType :'json', 
				success : function(data) { 
					
					if(entcaptxt===data[0] )
					{
						getAppState();
					}
					else
					{
						document.getElementById("nextText").innerHTML="Captcha is not matched, please try again";
						alert("Captcha is not matched, please try again");
						changeCaptcha();
						document.forms["authentication"]["llappln"].value =""; 
						document.forms["authentication"]["entcaptxt"].value =""; 
					    document.forms["authentication"]["llappln"].focus();
					}
				
				},
				error : function(error) {
					//called when there is an error
					alert("Captcha is not matched, please try again");
					changeCaptcha();
					document.forms["authentication"]["llappln"].value =""; 
					document.forms["authentication"]["entcaptxt"].value =""; 
				    document.forms["authentication"]["llappln"].focus();
				}
			});
		} catch (err) {
			//alert("error");
		}
	}
}

  function dobValidation()
  {
		var spclChars = "!@#$%^&*()+=[]\\\';,./{}|\":<>?"; // specify special characters 
		var content = document.forms["authentication"]["dob"].value;
	  /*Date of Birth - null/empty validation  */
		if (document.forms["authentication"]["dob"].value == null
				|| document.forms["authentication"]["dob"].value == "") {
			alert("Date of Birth must not empty");
			document.forms["authentication"]["dob"].focus();
			return false;
		}
		/*Date pattern validation */
		else if (!document.forms["authentication"]["dob"].value
				.match(/^(0[1-9]|[12][0-9]|3[01])[\ \-.](?:(0[1-9]|1[012])[\ \-.](19|20)[0-9]{2})$/)) {
			alert("Date Of Birth Should be in dd-mm-yyyy format only");

			document.forms["authentication"]["dob"].focus();
			return false;
		}
		/*Date of Birth - Age(must be >= 16yrs) validation  */
		else if (document.forms["authentication"]["dob"].value != null
				&& document.forms["authentication"]["dob"].value != ""
				&& getAge(document.forms["authentication"]["dob"].value) < 16) {
			alert("Age must be greater than or equal to 16 years only");

			document.forms["authentication"]["dob"].focus();
			return false;
		}
		else if(content.length>0)
		{
			for (var i = 0; i < content.length; i++) {
				if (spclChars.indexOf(content.charAt(i)) != -1) {
					alert("Special characters are not allowed.");
	
					document.forms["authentication"]["dob"].focus();
					return false;
				}
			}
		}
	  
	    if(document.forms["authentication"]["pwd"].value.length>5)
	    {
	    	$("#progressbar li").eq(1).addClass("active");
	    	$("#progressbar li").eq(0).addClass("process");

	    	$("#capphto1").attr('disabled',true);

	    }
  }
